def get_line_profile(event):
    """คืนค่า (display_name, picture_url) รองรับทั้งห้องส่วนตัว กลุ่ม และรูม"""
    display_name, picture_url = None, None
    try:
        if event.source.type == "user" and event.source.user_id:
            prof = line_bot_api.get_profile(event.source.user_id)
            display_name = prof.display_name
            picture_url = getattr(prof, "picture_url", None)
        elif event.source.type == "group" and event.source.group_id and event.source.user_id:
            prof = line_bot_api.get_group_member_profile(event.source.group_id, event.source.user_id)
            display_name = prof.display_name
            picture_url = getattr(prof, "picture_url", None)
        elif event.source.type == "room" and event.source.room_id and event.source.user_id:
            prof = line_bot_api.get_room_member_profile(event.source.room_id, event.source.user_id)
            display_name = prof.display_name
            picture_url = getattr(prof, "picture_url", None)
    except Exception as e:
        print("⚠️ get_line_profile error:", e)
    if not display_name:
        display_name = "ผู้ใช้"
    return display_name, picture_url

def format_thb(n: int) -> str:
    try:
        return f"{int(n):,}"
    except Exception:
        return str(n)

def private_reply(event, messages, group_ack: bool = True):
    """
    ส่งข้อความ/เฟล็กซ์แบบส่วนตัวถึง user คนนั้น
    - ถ้าเป็นแชท 1:1 => reply ปกติ
    - ถ้าอยู่ใน group/room => push ไปหา user โดยตรง แล้ว (ถ้าตั้ง group_ack=True) จะ reply ยืนยันสั้นๆในห้อง
    """
    try:
        if not isinstance(messages, list):
            messages = [messages]

        if event.source.type == "user":
            line_bot_api.reply_message(event.reply_token, messages)
            return

        user_id = event.source.user_id
        if not user_id:
            if group_ack:
                line_bot_api.reply_message(event.reply_token, TextSendMessage("⚠️ ไม่สามารถส่งส่วนตัวได้"))
            return

        line_bot_api.push_message(user_id, messages)

        if group_ack:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("✅ ส่งข้อมูลให้คุณทางแชทส่วนตัวแล้ว"))
    except Exception as e:
        print("⚠️ private_reply error:", e)
        line_bot_api.reply_message(
            event.reply_token,
            TextSendMessage("⚠️ ส่งส่วนตัวไม่สำเร็จ กรุณาแอดบอทเป็นเพื่อนและทักแชทส่วนตัวก่อนนะครับ")
        )

                 "paddingAll": "12px", "contents": [header, {"type": "separator", "margin": "md"}] + rows}
            ]
        }
    }
    return bubble

def build_credit_table_flex(all_members):
    """
    all_members: list of tuple (rid, name, credit)
    สร้าง FlexSendMessage แบบ carousel (แบ่งหน้า หน้า/บับเบิลละ <= 15 แถว)
    """
    PAGE_SIZE = 15
    records = [{"rid": rid, "name": name or "ผู้ใช้", "credit": int(credit or 0)}
               for (rid, name, credit) in all_members]
    if not records:
        return TextSendMessage("ℹ️ ยังไม่มีสมาชิก")

    pages = [records[i:i+PAGE_SIZE] for i in range(0, len(records), PAGE_SIZE)]
    total_pages = len(pages)

    bubbles = [build_credit_table_bubble(page, idx, total_pages) for idx, page in enumerate(pages, start=1)]
    bubbles = bubbles[:10]  # สูงสุด 10 บับเบิลต่อข้อความ

    contents = {"type": "carousel", "contents": bubbles} if len(bubbles) > 1 else bubbles[0]
    return FlexSendMessage(alt_text="ตารางเครดิตสมาชิก", contents=contents)

# ================== PARSERS ==================
LOW_ALIASES = {"ต", "ย", "ต่ำ", "low", "l"}
HIGH_ALIASES = {"ส", "สูง", "high", "h"}

def parse_bet_text(text: str):
    """รองรับ: ต/100, ย100, ส200, ต่ำ100, สูง/300"""
    t = text.strip().lower().replace(" ", "")
    m = re.match(r"([^\d/]+)[/]?(\d+)$", t)
    if m:
        alias = m.group(1)
        amount = int(m.group(2))
        if alias in LOW_ALIASES:
            return "ต่ำ", amount
        if alias in HIGH_ALIASES:
            return "สูง", amount
    m2 = re.match(r"(ต่ำ|สูง)[/]?(\d+)$", t)
    if m2:
        side = m2.group(1)
        amount = int(m2.group(2))
        return side, amount
    return None, None

def parse_admin_credit_command(text: str):
    """
    รองรับรูปแบบ:
    ใหม่:   $+ <id> <amount>
    เดิม:   เติม 500 @3 | ลบ 200 @7
            เติมเครดิต 1000 @2 | ลบเครดิต 300 @5
            addcredit 500 @3 | subcredit 300 @5
    คืนค่า: (op, amount, rid) หรือ (None, None, None)
    """
    t = text.strip()

    # ----- รูปแบบใหม่: $+ <id> <amount> -----
    m_new_add = re.match(r"^\$\+\s*(\d+)\s+(\d+)$", t)
    if m_new_add:
        rid = int(m_new_add.group(1))
        amount = int(m_new_add.group(2))
        return "add", amount, rid

    # ===== รูปแบบเดิม ๆ =====
    tl = t.lower()
    if tl.startswith(("เติมเครดิต", "เติม ", "addcredit")):
        op = "add"
        rest = tl.replace("เติมเครดิต", "", 1).replace("เติม ", "", 1).replace("addcredit", "", 1).strip()
    elif tl.startswith(("ลบเครดิต", "ลบ ", "subcredit")):
        op = "sub"
        rest = tl.replace("ลบเครดิต", "", 1).replace("ลบ ", "", 1).replace("subcredit", "", 1).strip()
    else:
        return None, None, None

    m_old = re.match(r"(\d+)\s*@(\d+)$", rest)
    if not m_old:
        return None, None, None
    amount = int(m_old.group(1))
    rid = int(m_old.group(2))
    return op, amount, rid

# ================== SETTLEMENT ==================
def settle_round(round_id: int, code: str):
    """
    สรุปผลรอบตามรหัส:
    ส=สูงชนะหัก10% , ต=ต่ำชนะหัก10% ,
    ตส=ต่ำเสีย10% • สูงเสมอ ,
    ตจ=สูงเสีย10% • ต่ำเสมอ ,
    ก=กลาง5% , ม=เสมอ(ไม่หัก%)
    """
    if round_id not in rounds:
        return {"text": "ไม่พบรอบดังกล่าว", "items": [], "total_plus": 0, "total_minus": 0, "net": 0}
    r = rounds[round_id]
    if r.get("settled"):
        return {"text": f"รอบ #{round_id} ถูกสรุปไปแล้ว", "items": [], "total_plus": 0, "total_minus": 0, "net": 0}

    bets = r.get("bets", {})
    if not bets:
        r["settled"] = True
        r["result_code"] = code
        return {"text": f"รอบ #{round_id} ไม่มีบิลให้สรุป", "items": [], "total_plus": 0, "total_minus": 0, "net": 0}

    def apply_rule(side, amt, code):
        # ชนะ = +amt*(1-fee), แพ้เต็ม = -amt, แพ้ x% = -amt*x, เสมอ = 0
        if code == "ส":
            return ((amt * (1 - 0.10), "ชนะ(หัก10%)") if side == "สูง" else (-amt, "แพ้เต็ม"))
        if code == "ต":
            return ((amt * (1 - 0.10), "ชนะ(หัก10%)") if side == "ต่ำ" else (-amt, "แพ้เต็ม"))
        if code == "ตส":  # ต่ำเสีย10% • สูงเสมอ
            return ((-amt * 0.10, "แพ้10%") if side == "ต่ำ" else (0, "เสมอ"))
        if code == "ตจ":  # สูงเสีย10% • ต่ำเสมอ
            return ((-amt * 0.10, "แพ้10%") if side == "สูง" else (0, "เสมอ"))
        if code == "ก":   # กลาง 5% ทั้งสองฝั่งเสีย5%
            return (-amt * 0.05, "เสีย5%")
        if code == "ม":   # เสมอ ไม่หัก%
            return (0, "เสมอ")
        return (0, "ไม่รู้จักรหัส")

    lines = []
    total_plus = 0.0
    total_minus = 0.0

    for uid, bet in bets.items():
        if bet["status"] != "active":
            continue
        side = bet["side"]
        amt = bet["amount"]
        delta, verdict = apply_rule(side, amt, code)

        # อัปเดตเครดิตสมาชิก (ปัดเป็น int)
        m = members.get(uid)
        if m:
            m["credit"] = m.get("credit", 0) + int(round(delta))

        # อัปเดตสถานะบิล
        bet["status"] = "settled"
        bet["payout"] = int(round(delta))
        bet["result"] = verdict

        rid = members.get(uid, {}).get("id", "-")
        name = members.get(uid, {}).get("name", "ผู้ใช้")
        lines.append({
            "rid": rid,
            "name": name,
            "side": side,
            "amount": amt,
            "payout": int(round(delta)),
            "result": verdict
        })

        if delta >= 0:
            total_plus += delta
        else:
            total_minus += (-delta)

    r["settled"] = True
    r["result_code"] = code

    net = int(round(total_plus - total_minus))
    return {
        "text": f"สรุปยอด รอบ #{round_id} ({code}) เสร็จสิ้น",
        "items": lines,
        "total_plus": int(round(total_plus)),
        "total_minus": int(round(total_minus)),
        "net": net
    }

@handler.add(MessageEvent, message=TextMessage)
def handle_message(event):
    global next_running_id, next_bet_id, current_round, active_bets, rounds, pending_settlements

    user_id = event.source.user_id
    text = event.message.text.strip()

    # ----- แสดง userId (ส่งส่วนตัว) -----
    if text.lower() == "myid":
        private_reply(event, TextSendMessage(text=f"📌 userId ของคุณคือ:\n{user_id}"))
        return

    # ----- สมัครสมาชิก: add (ส่งการ์ดส่วนตัว) -----
    if text.lower().startswith("add"):
        display_name, picture_url = get_line_profile(event)
        if user_id in members:
            private_reply(event, build_status_flex(members[user_id]))
            return
        members[user_id] = {
            "role": "ลูกค้า",
            "name": display_name,
            "picture_url": picture_url,
            "id": next_running_id,
            "registered_at": datetime.now(timezone.utc),
            "credit": 0,
            "capital": 0,
        }
        next_running_id += 1
        private_reply(event, build_status_flex(members[user_id]))
        return

    # ----- กด C เพื่อดูการ์ดสถานะ (ส่งส่วนตัว) -----
    if text.lower() == "c":
        if user_id not in members:
            # แจ้งสาธารณะได้เพราะไม่ใช่ข้อมูลส่วนตัว
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⚠️ กรุณาสมัครสมาชิกก่อน โดยพิมพ์: add"))
            return

        display_name, picture_url = get_line_profile(event)
        members[user_id]["name"] = display_name or members[user_id]["name"]
        if picture_url:
            members[user_id]["picture_url"] = picture_url

        bet_for_card = None
        rid_now = current_round.get("id", 0)
        if rid_now and rid_now in rounds:
            bet_for_card = rounds[rid_now].get("bets", {}).get(user_id)
            if bet_for_card and bet_for_card.get("status") != "active":
                bet_for_card = None

        private_reply(
            event,
            build_status_flex(
                members[user_id],
                bet=bet_for_card,
                round_state=current_round.get("state"),
                round_id=rid_now
            ),
            group_ack=True
        )
        return

    # ----- ถ้ายังไม่สมัคร -> บังคับสมัครก่อน -----
    if user_id not in members:
        line_bot_api.reply_message(event.reply_token, TextSendMessage("⚠️ กรุณาสมัครสมาชิกก่อน โดยพิมพ์: add"))
        return

    upper = text.strip().upper()

    # ======= คำสั่งแอดมิน: เปิดรอบ(Oชื่อ)/หยุด(E)/ปิด(X) =======

    # เปิดรอบด้วยรูปแบบ O<ชื่อรอบ> เช่น Oนครสวรรค์
    if upper.startswith("O"):
        if not is_admin(user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return

        # บังคับต้องมีชื่อรอบ
        round_name = text[1:].strip()
        if not round_name:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ ต้องระบุชื่อรอบก่อนเปิด เช่น: Oนครสวรรค์"))
            return

        # resume จาก paused ของรอบเดิม
        if current_round["state"] == "paused":
            rid = current_round["id"]
            current_round["state"] = "open"
            current_round["name"] = round_name
            rounds[rid]["state"] = "open"
            rounds[rid]["name"] = round_name
            line_bot_api.reply_message(event.reply_token, build_round_open_flex(round_name))
            return
        elif current_round["state"] == "open":
            line_bot_api.reply_message(event.reply_token, TextSendMessage(f"🔓 รอบ #{current_round['id']} เปิดอยู่แล้ว"))
            return
        elif current_round["state"] == "closed":
            last_rid = current_round["id"]
            if last_rid in rounds and not rounds[last_rid].get("settled", False):
                line_bot_api.reply_message(
                    event.reply_token,
                    TextSendMessage(f"ℹ️ รอบ #{last_rid} ยังไม่สรุปผล กรุณาสรุป (S<รหัส>) และยืนยันก่อนเปิดรอบใหม่")
                )
                return

        # เปิดรอบใหม่เมื่อ idle หรือ closed(และสรุปแล้ว)
        if current_round["state"] in {"idle", "closed"}:
            current_round["id"] += 1
            rid = current_round["id"]
            current_round["state"] = "open"
            current_round["opened_at"] = datetime.now(TZ_BKK)
            current_round["closed_at"] = None
            current_round["name"] = round_name

            active_bets.clear()
            rounds[rid] = {
                "state": "open",
                "opened_at": current_round["opened_at"],
                "closed_at": None,
                "bets": {},
                "settled": False,
                "result_code": None,
                "name": round_name,
            }
            line_bot_api.reply_message(event.reply_token, build_round_open_flex(round_name))
            return

    # ======= คำสั่งหยุด E =======
    if upper == "E":
        if current_round["state"] == "open":
            current_round["state"] = "paused"
            rid = current_round["id"]
            rounds[rid]["state"] = "paused"
            line_bot_api.reply_message(event.reply_token, TextSendMessage(f"⏸️ หยุดรับแทงชั่วคราว รอบ #{rid}"))
        else:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("ℹ️ ยังไม่มีรอบที่เปิดอยู่ให้หยุด"))
        return

    # ======= คำสั่งปิด X =======
    if upper == "X":
        if current_round["state"] in {"open", "paused"}:
            current_round["state"] = "closed"
            current_round["closed_at"] = datetime.now(TZ_BKK)
            rid = current_round["id"]
            rounds[rid]["state"] = "closed"
            rounds[rid]["closed_at"] = current_round["closed_at"]
            line_bot_api.reply_message(event.reply_token, TextSendMessage(f"🚫 ปิดรอบ #{rid} แล้ว"))
        else:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("ℹ️ ยังไม่มีรอบที่กำลังเปิดให้ปิด"))
        return

    # ======= คำสั่งสรุปผล S<รหัส> (เฉพาะแอดมิน) =======
    if text.startswith(("S", "s")):
        if not is_admin(user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return
        code = text[1:].strip()
        rid = current_round["id"]
        if rid == 0 or rounds.get(rid, {}).get("state") != "closed":
            line_bot_api.reply_message(event.reply_token, TextSendMessage("ℹ️ ต้องปิดรอบก่อน (X) จึงจะสรุปได้"))
            return

        pending_settlements[user_id] = {"round_id": rid, "code": code}
        admin_name, _ = get_line_profile(event)
        # ส่งยืนยันสรุปแบบสาธารณะได้ เพราะไม่ใช่ข้อมูลส่วนตัว
        line_bot_api.reply_message(event.reply_token, build_settle_confirm_flex(rid, code, admin_name))
        return

    # ======= เติม/ลบเครดิต (เฉพาะแอดมิน) =======
    op, amount, rid = parse_admin_credit_command(text)
    if op:
        if not is_admin(user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return
        target_uid, target_member = find_member_by_running_id(rid)
        if not target_member:
            line_bot_api.reply_message(event.reply_token, TextSendMessage(f"ไม่พบสมาชิก ID {rid}"))
            return
        if op == "add":
            target_member["credit"] = target_member.get("credit", 0) + amount
            msg = f"✅ เติมเครดิตให้ ID {rid} จำนวน {format_thb(amount)} บาท\nเครดิตรวม: {format_thb(target_member['credit'])}"
        else:
            target_member["credit"] = max(0, target_member.get("credit", 0) - amount)
            msg = f"✅ ลบเครดิตของ ID {rid} จำนวน {format_thb(amount)} บาท\nเครดิตรวม: {format_thb(target_member['credit'])}"
        line_bot_api.reply_message(event.reply_token, TextSendMessage(msg))
        return

    # ----- ตารางเครดิตสมาชิกทั้งหมด: call (ส่งส่วนตัวแก่ผู้สั่ง) -----
    if text.lower() == "call":
        # อนุญาตเฉพาะแอดมิน เพื่อปกป้องข้อมูลสมาชิก
        if not is_admin(user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return

        all_list = []
        for uid, m in members.items():
            rid_m = m.get("id", 0)
            name_m = m.get("name", "ผู้ใช้")
            credit_m = m.get("credit", 0)
            all_list.append((rid_m, name_m, credit_m))
        all_list.sort(key=lambda x: x[0])

        if not all_list:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("ℹ️ ยังไม่มีสมาชิกที่สมัครเข้ามา"))
            return

        flex_msg = build_credit_table_flex(all_list)
        private_reply(event, flex_msg, group_ack=True)
        return

    # ======= รับบิลลูกค้า (ยืนยันส่งส่วนตัว) =======
    side, bet_amount = parse_bet_text(text)
    if side and bet_amount and bet_amount > 0:
        if current_round["state"] != "open":
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ ยังไม่เปิดรอบ หรือกำลังหยุดรับแทง"))
            return

        user_credit = members[user_id].get("credit", 0)
        if user_credit <= 0:
            private_reply(event, TextSendMessage("⛔ เครดิตไม่พอ กรุณาเติมเครดิตก่อนทำรายการ"))
            return
        if user_credit < bet_amount:
            private_reply(event, TextSendMessage(
                f"⛔ เครดิตของคุณมี {format_thb(user_credit)} บาท ไม่พอสำหรับแทง {format_thb(bet_amount)} บาท"
            ))
            return

        if user_id in active_bets:
            existing = active_bets[user_id]
            if existing["round_id"] == current_round["id"] and existing["status"] == "active":
                private_reply(event, TextSendMessage("⚠️ คุณลงเดิมพันในรอบนี้ไปแล้ว ไม่สามารถลงซ้ำได้"))
                return

        bet = {
            "bet_id": next_bet_id,
            "round_id": current_round["id"],
            "user_id": user_id,
            "side": side,
            "amount": bet_amount,
            "status": "active",
            "created_at": datetime.now(TZ_BKK)
        }
        active_bets[user_id] = bet
        next_bet_id += 1

        rid_now = current_round["id"]
        if rid_now not in rounds:
            rounds[rid_now] = {"state": "open", "opened_at": datetime.now(TZ_BKK),
                               "closed_at": None, "bets": {}, "settled": False, "result_code": None, "name": current_round.get("name")}
        rounds[rid_now]["bets"][user_id] = bet

        private_reply(event, build_bet_flex(members[user_id], bet))
        return

@handler.add(PostbackEvent)
def handle_postback(event):
    global pending_settlements

    data = event.postback.data or ""
    params = dict(urllib.parse.parse_qsl(data))
    action = params.get("action")

    if action == "cancel_bet":
        pb_user = params.get("user_id")
        bet_id = int(params.get("bet_id", "0") or "0")

        if pb_user != event.source.user_id:
            private_reply(event, TextSendMessage("⛔ ไม่สามารถยกเลิกบิลของผู้อื่นได้"), group_ack=False)
            return

        bet = active_bets.get(pb_user)
        if not bet or bet["bet_id"] != bet_id or bet["status"] != "active":
            private_reply(event, TextSendMessage("ไม่พบบิลที่ใช้งานหรือถูกยกเลิกไปแล้ว"), group_ack=False)
            return

        if current_round["state"] == "closed" or bet["round_id"] != current_round["id"]:
            private_reply(event, TextSendMessage("⛔ ไม่สามารถยกเลิกบิลหลังปิดรอบ"), group_ack=False)
            return

        bet["status"] = "cancelled"
        private_reply(event, TextSendMessage(f"✅ ยกเลิกการเดิมพัน #{bet_id} เรียบร้อย"), group_ack=False)
        return

    if action == "confirm_settle":
        if not is_admin(event.source.user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return

        pend = pending_settlements.get(event.source.user_id)
        if not pend:
            line_bot_api.reply_message(event.reply_token, TextSendMessage("ℹ️ ไม่พอข้อมูลการสรุป กรุณาเริ่มใหม่"))
            return

        rid = int(params.get("round_id", pend["round_id"]))
        code = params.get("code", pend["code"])
        result = settle_round(rid, code)

        # ลบสถานะรอ
        pending_settlements.pop(event.source.user_id, None)

        # ส่งสรุปผลให้แอดมินแบบส่วนตัว (และแจ้งยืนยันในห้อง)
    flex_summary = build_settlement_flex(
        round_id=rid,
        code=code,
        items=result["items"],
        total_plus=result["total_plus"],
        total_minus=result["total_minus"],
        net=result["net"]
    )

    round_name = rounds[rid].get("name", f"รอบ {rid}")
    flex_score = build_score_flex(round_name, code)

    line_bot_api.reply_message(event.reply_token, [flex_summary, flex_score])
    return


    if action == "cancel_settle":
        if not is_admin(event.source.user_id):
            line_bot_api.reply_message(event.reply_token, TextSendMessage("⛔ เฉพาะแอดมินเท่านั้น"))
            return
        pending_settlements.pop(event.source.user_id, None)
        line_bot_api.reply_message(event.reply_token, TextSendMessage("✅ ยกเลิกการสรุปผลแล้ว"))
        return

    # default
    line_bot_api.reply_message(event.reply_token, TextSendMessage("⚙️ ดำเนินการไม่สำเร็จ"))
