def parse_bet(text):
    m = re.match(r"^([ลสยต])\s*([0-9]+)$", text, re.IGNORECASE)
    if not m: return None
    ch = m.group(1).lower()
    amount = int(m.group(2))
    side = "HI" if ch in ("ล", "ส") else "LO"
    return {"side": side, "amount": amount}

def get_user_bet(state, uid): return state["bet_index"].get(uid)
def user_stake_this_round(state, uid): return get_user_bet(state, uid)["amount"] if get_user_bet(state, uid) else 0
def user_fund(state, uid): return state["funds"].get(uid, 0)
def user_fund_remain(state, uid):
    u = users.get(uid)
    if not u:
        return 0
    credit_total = u.get("credit", 0)
    return max(credit_total - user_stake_this_round(state, uid), 0)

def can_bet(state, uid, side, amount):
    if state["phase"] != "OPEN": return (False, "ยังไม่เปิดรอบ")

    existing = get_user_bet(state, uid)
    if existing:
        exist_side_th = "สูง" if existing["side"] == "HI" else "ต่ำ"
        if side != existing["side"]:
            return (False, f"❌ ห้ามแทงสวน — คุณมีบิลเดิม: {exist_side_th} {fmt(existing['amount'])}  (พิมพ์ X เพื่อยกเลิกก่อน)")
        else:
            return (False, f"❌ จำกัด 1 บิล/รอบ — คุณมีบิล {exist_side_th} {fmt(existing['amount'])} อยู่แล้ว  (พิมพ์ X เพื่อยกเลิกก่อน)")

    if amount < MIN_BET: return (False, f"ขั้นต่ำ {MIN_BET}")
    if amount > MAX_BET: return (False, f"สูงสุด {MAX_BET}")

    remain = user_fund_remain(state, uid)
    if remain < amount: return (False, f"ทุนคงเหลือไม่พอ (มี {fmt(remain)})")

    if amount > USER_SIDE_CAP[side]:
        return (False, f"ฝั่ง{'สูง' if side=='HI' else 'ต่ำ'} ต่อคนเกิน {fmt(USER_SIDE_CAP[side])}")
    if state["totals"][side] + amount > SIDE_CAP[side]:
        return (False, f"ฝั่ง{'สูง' if side=='HI' else 'ต่ำ'} เต็ม {fmt(SIDE_CAP[side])}")
    if state["totals"]["HI"] + state["totals"]["LO"] + amount > ROUND_CAP:
        return (False, f"รอบนี้เต็ม {fmt(ROUND_CAP)}")

    return (True, "")

def first_mentioned_uid(event):
    """
    คืน userId ตัวแรกที่ถูกแท็กในข้อความ (ถ้าแท็ก @all จะข้าม)
    ใช้ได้เฉพาะในกลุ่ม/ห้อง
    """
    try:
        m = getattr(event.message, "mention", None)
        if not m: return None
        for me in (getattr(m, "mentionees", None) or []):
            uid = getattr(me, "user_id", None) or getattr(me, "userId", None)
            if uid and str(uid).lower() != "all":
                return uid
    except Exception:
        pass
    return None

def settle_by_code(st, code):
    acc = {}  # uid -> row
    def add(uid, name, stake, pnl):
        row = acc.get(uid, {"uid": uid, "name": name, "stake": 0, "pnl": 0})
        row["stake"] += stake
        row["pnl"] += pnl
        acc[uid] = row

    d = RESULT_DEFS.get(code)

    # จาว (คืนเต็ม)
    if not d or d.get("special") == "DRAW_0":
        for b in st["bet_index"].values():
            add(b["uid"], b["name"], b["amount"], 0)
        return list(acc.values()), "ผล: จาว (คืนเต็ม)"

    # กลาง (คืนเงิน หัก 5%)
    if d.get("special") == "MIDDLE_5":
        for b in st["bet_index"].values():
            add(b["uid"], b["name"], b["amount"], -round(b["amount"]*0.05))
        return list(acc.values()), "ผล: กลาง (คืนเงิน หัก 5%)"

        # ต่ำเสีย 10% / สูงเสีย 5%
    if d.get("special") == "LOW_LOSE_10":
        for b in st["bet_index"].values():
            if b["side"] == "LO":
                pnl = -round(b["amount"] * 0.10)   # ต่ำเสีย 10%
            elif b["side"] == "HI":
                pnl = -round(b["amount"] * 0.05)   # สูงเสีย 5%
            else:
                pnl = 0
            add(b["uid"], b["name"], b["amount"], pnl)
        return list(acc.values()), "ผล: ต่ำเสีย 10% / สูงเสีย 5%"

    # ต่ำไม่เสีย / สูงเสีย 10%
    elif d.get("special") == "LOW_NOLOSE_HIGH_LOSE10":
        for b in st["bet_index"].values():
            pnl = 0 if b["side"] == "LO" else -round(b["amount"] * 0.10)
            add(b["uid"], b["name"], b["amount"], pnl)
        return list(acc.values()), "ผล: ต่ำไม่เสีย / สูงเสีย 10%"


    # ปกติ: มีฝั่งชนะ/แพ้
    win, fee, lose_pct = d["winner"], d.get("winnerFee", 0), d.get("loserLose", 1.0)
    for b in st["bet_index"].values():
        pnl = round(b["amount"]*(1-fee)) if b["side"] == win else -round(b["amount"]*lose_pct)
        add(b["uid"], b["name"], b["amount"], pnl)
    return list(acc.values()), f"ผล: {d['label']}"

class RateLimiter:
    def __init__(self): self._buckets = {}
    def allow(self, key: str, limit: int, period: int) -> bool:
        now = time.time()
        dq = self._buckets.get(key)
        if dq is None:
            dq = deque(); self._buckets[key] = dq
        while dq and (now - dq[0]) > period:
            dq.popleft()
        if len(dq) >= limit:
            return False
        dq.append(now)
        return True

def webhook():
    if request.content_length and request.content_length > MAX_BODY_BYTES:
        time.sleep(0.2)
        return "payload too large", 413

    if REQUIRE_LINE_UA:
        ua = request.headers.get("User-Agent", "")
        if not _safe_is_line_ua(ua):
            time.sleep(0.2)
            return "forbidden ua", 403

    ip = _client_ip()
    if not rl.allow(f"ip:{ip}", RL_IP_LIMIT, RL_IP_PERIOD):
        time.sleep(0.2)
        return "too many", 429

    body = request.get_data(as_text=True)
    ts_hdr = request.headers.get("X-Line-Request-Timestamp", "").strip()
    if ts_hdr.isdigit():
        try:
            tsv = int(ts_hdr)
            if tsv > 10**12: tsv = int(tsv / 1000)
            drift = abs(_now() - tsv)
            if drift > WEBHOOK_DRIFT_SEC:
                return "stale request", 401
        except Exception:
            pass

    sig = request.headers.get("X-Line-Signature", "")
    expected = base64.b64encode(hmac_new(CHANNEL_SECRET.encode(), body.encode(), sha256).digest()).decode()
    if not sig or not compare_digest(sig, expected):
        return "signature error", 400

    try:
        handler.handle(body, sig)
    except InvalidSignatureError:
        return "signature error", 400
    except Exception as e:
        app.logger.exception("err: %s", e)
        return "error", 500
    return "OK"

@handler.add(MessageEvent, message=TextMessage)
def on_message(event: MessageEvent):
    global nextCustomerId

    if not in_group_or_room(event.source):
        return

    uid = event.source.user_id
    gid = getattr(event.source, "group_id", None)
    key = room_key(event.source)

    # Group allow/deny/ban
    if gid:
        if gid in BANNED_GROUPS or gid in DENY_GROUP_IDS:
            return
        if not is_allowed_group(gid):
            return
        if _locked_group(gid) and not is_admin(uid):
            if not _notice_throttled(uid):
                safe_reply(event, TextSendMessage("ระบบ: กลุ่มกำลังล็อกดาวน์ชั่วคราว ติดต่อแอดมินเพื่อปลดล็อก"))
            return

    # user ban/mute
    if uid in BANNED_UIDS:
        return
    if MUTED_UNTIL.get(uid, 0) > _now():
        if not _notice_throttled(uid):
            safe_reply(event, TextSendMessage("ระบบ: คุณถูกจำกัดการส่งข้อความชั่วคราว (anti-spam)"))
        return

    # rate limit room/user
    if not rl.allow(f"room:{key}", RL_ROOM_BURST_LIMIT, RL_ROOM_BURST_PERIOD):
        return
    if not rl.allow(f"uid:{uid}:burst", RL_UID_BURST_LIMIT, RL_UID_BURST_PERIOD) or \
       not rl.allow(f"uid:{uid}:day", RL_UID_DAILY_LIMIT, RL_UID_DAILY_PERIOD):
        STRIKES[uid] = STRIKES.get(uid, 0) + 1
        if STRIKES[uid] >= 3:
            MUTED_UNTIL[uid] = _now() + MUTE_SECONDS_DEFAULT
            STRIKES[uid] = 0
            if not _notice_throttled(uid):
                safe_reply(event, TextSendMessage(f"ระบบ: มิวท์ {MUTE_SECONDS_DEFAULT} วินาที เนื่องจากข้อความถี่ผิดปกติ"))
        else:
            if not _notice_throttled(uid):
                safe_reply(event, TextSendMessage("ระบบ: ข้อความถี่เกินกำหนด ช่วยเว้นช่วงหน่อยนะ"))
        return

    # cache for unsend monitor
    msgCache[event.message.id] = {"text": (event.message.text or ""), "ts": time.time()}
    now = time.time()
    if len(msgCache) > 4000:
        for k, v in list(msgCache.items()):
            if now - v["ts"] > CACHE_TTL_SEC: msgCache.pop(k, None)

    # state
    if key not in rooms: rooms[key] = start_state()
    st = rooms[key]

    text = (event.message.text or "").strip()

    # ===== Admin: add/del with @mention or Uxxxxxxxx + optional PIN =====
    if re.match(r"^admin\s+add\b", text, re.IGNORECASE):
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        target_uid = first_mentioned_uid(event)
        if not target_uid:
            m = re.match(r"^admin\s+add\s+([Uu][0-9a-f]{32})(.*)$", text)
            if m: target_uid = m.group(1)
        if not target_uid:
            safe_reply(event, TextSendMessage("โปรดแท็กผู้ใช้ หรือระบุ userId ที่ขึ้นต้นด้วย U...")); return
        pin = _admin_auth_pin(text)
        if ADMIN_PIN and not compare_digest(pin, ADMIN_PIN):
            safe_reply(event, TextSendMessage("PIN ไม่ถูกต้อง")); return
        add_admin(target_uid)
        safe_reply(event, TextSendMessage("เพิ่มแอดมินสำเร็จ ✓")); return

    if re.match(r"^admin\s+del\b", text, re.IGNORECASE):
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        target_uid = first_mentioned_uid(event)
        if not target_uid:
            m = re.match(r"^admin\s+del\s+([Uu][0-9a-f]{32})(.*)$", text)
            if m: target_uid = m.group(1)
        if not target_uid:
            safe_reply(event, TextSendMessage("โปรดแท็กผู้ใช้ หรือระบุ userId ที่ขึ้นต้นด้วย U...")); return
        pin = _admin_auth_pin(text)
        if ADMIN_PIN and not compare_digest(pin, ADMIN_PIN):
            safe_reply(event, TextSendMessage("PIN ไม่ถูกต้อง")); return
        try:
            ADMIN_IDS.remove(target_uid)
            safe_reply(event, TextSendMessage("ลบแอดมินสำเร็จ ✓"))
        except ValueError:
            safe_reply(event, TextSendMessage("ไม่พบไอดีนี้ในรายชื่อแอดมิน"))
        return

    # ==== ดูตารางรายการเดิมพัน (cm — เฉพาะกลุ่มหลังบ้าน) ====
    if text.lower() == "cm":

                })

                    total_hi += b["amount"]
                else:
                    total_lo += b["amount"]

            safe_reply(event, TextSendMessage("(ยังไม่มีบิลในระบบ)"))
            return

        rows = [
            {"type":"box","layout":"horizontal","contents":[
                {"type":"text","text":"ผู้เล่น","flex":5,"size":"sm","weight":"bold"},
                {"type":"text","text":"สูง/ต่ำ","flex":2,"size":"sm","align":"end","weight":"bold"},

                {"type":"text","text":"รอบ","flex":2,"size":"sm","align":"end","weight":"bold"},
            ]},

        ]

        for b in all_bets:

                {"type":"text","text":"สูง" if b["side"]=="HI" else "ต่ำ",

        return

    # ===== Moderator: ban/mute/unban/unmute (รองรับ @แท็ก) =====
    m_cmd = re.match(r"^(ban|unban|mute|unmute)\b(?:\s+(.*))?$", text, re.IGNORECASE)
    if m_cmd:
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน"))
            return
        cmd = m_cmd.group(1).lower()
        args = (m_cmd.group(2) or "").strip()
        target_uid = first_mentioned_uid(event)
        sec = None
        if not target_uid:
            m_uid = re.match(r"^(U[0-9a-f]{32})\b(?:\s+(\d+))?$", args, re.IGNORECASE)
            if m_uid:
                target_uid = m_uid.group(1)
                sec = int(m_uid.group(2) or MUTE_SECONDS_DEFAULT) if cmd == "mute" else None
            else:
                m_at = re.match(r"^@(.+?)(?:\s+(\d+))?$", args)
                if m_at:
                    safe_reply(event, TextSendMessage("โปรดแท็กผู้ใช้จาก UI ของ LINE (ชื่อเป็นลิงก์สีน้ำเงิน)"))
                    return
        if not target_uid:
            safe_reply(event, TextSendMessage("รูปแบบ: mute @ผู้ใช้ [วินาที] / unmute @ผู้ใช้ / ban @ผู้ใช้ / unban @ผู้ใช้"))
            return
        if cmd == "mute":
            if sec is None:
                m_sec = re.search(r"\b(\d+)\b$", args) if args else None
                sec = int(m_sec.group(1)) if m_sec else MUTE_SECONDS_DEFAULT
            MUTED_UNTIL[target_uid] = _now() + max(1, sec); safe_reply(event, TextSendMessage(f"มิวท์ {sec} วินาทีแล้ว")); return
        if cmd == "unmute":
            MUTED_UNTIL.pop(target_uid, None); safe_reply(event, TextSendMessage("ปลดมิวท์แล้ว")); return
        if cmd == "ban":
            BANNED_UIDS.add(target_uid); safe_reply(event, TextSendMessage("แบนผู้ใช้แล้ว")); return
        if cmd == "unban":
            BANNED_UIDS.discard(target_uid); safe_reply(event, TextSendMessage("ปลดแบนแล้ว")); return

    # ===== Admin: เช็ค UID ด้วยคำสั่ง uid @แท็ก =====
    if re.match(r"^uid\b", text, re.IGNORECASE):
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        target_uid = first_mentioned_uid(event)
        if not target_uid:
            safe_reply(event, TextSendMessage("โปรดใช้รูปแบบ: uid @ผู้ใช้ (ต้องแท็กจริง)")); return
        name, _ = get_profile_display(event.source, target_uid)
        safe_reply(event, TextSendMessage(f"UID ของ {name}: {target_uid}")); return

    # ==== CLEAR / RESET ====
    if re.match(r"^(clear|reset)\b", text, re.IGNORECASE):
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        if re.search(r"\ball\b", text, re.IGNORECASE):
            for rk in list(rooms.keys()):
                rooms[rk] = start_state()
            safe_reply(event, TextSendMessage("เคลียร์ทั้งระบบ (รอบ/ทุน) สำเร็จ ✓"))
            return
        else:
            rooms[key] = start_state()
            safe_reply(event, TextSendMessage("เคลียร์ห้องนี้ (รอบ/ทุน) สำเร็จ ✓"))
            return

# --- ส่วนของการเติมเครดิต ---
def process_credit_command(text, uid):
    if not is_admin(uid):  # ตรวจสอบว่าเป็นแอดมิน
        return "คำสั่งนี้ใช้ได้เฉพาะแอดมิน"

    # แยกชื่อและจำนวนเงิน
    m = re.match(r"^@([^\s]+)\s*/\s*(\d+)$", text)
    if not m:
        return "รูปแบบคำสั่งไม่ถูกต้อง"

    target_name = m.group(1)  # ชื่อลูกค้าที่ถูกแท็ก
    amt = int(m.group(2))  # จำนวนเงินที่เติม

    # ค้นหาผู้ใช้จากชื่อที่แท็ก
    target_user = None
    for user in users.values():
        if user["name"] == target_name:
            target_user = user
            break

    if not target_user:
        return f"ไม่พบผู้ใช้ {target_name}"

    # เติมเครดิตให้กับลูกค้า
    credit_before = target_user.get("credit", 0)
    target_user["credit"] = credit_before + amt

    return f"เติมเครดิตสำเร็จ ✓  {target_name} ({fmt(amt)} ฿)\nเครดิตก่อนหน้า: {fmt(credit_before)} ฿ ➜ {fmt(target_user['credit'])} ฿"
    # ==== เติม/ลบทุน+เครดิต ====
    m_add = re.match(r"^\$\+\s*(\d+)\s+(\d+)$", text)
    m_sub = re.match(r"^\$-\s*(\d+)\s+(\d+)$", text)
    if m_add or m_sub:
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        cid = int((m_add or m_sub).group(1)); amt = int((m_add or m_sub).group(2))
        target = get_user_by_cid(cid)
        if not target:
            safe_reply(event, TextSendMessage(f"ไม่พบ ID {cid}")); return

        tuid = target["uid"]
        fund_before = st["funds"].get(tuid, 0)
        credit_before = target.get("credit", 0)
        if m_add:
            target["credit"] = credit_before + amt
            st["funds"][tuid] = fund_before + amt
            op = "เติม"
        else:
            target["credit"] = max(credit_before - amt, 0)
            st["funds"][tuid] = max(fund_before - amt, 0)
            op = "ลบ"

        safe_reply(event, TextSendMessage(
            f"{op}สำเร็จ ✓  ID:{cid}\n"
            f"ทุนรอบนี้: {fmt(fund_before)} ➜ {fmt(st['funds'][tuid])} (±{fmt(amt)})\n"
            f"เครดิตสะสม: {fmt(credit_before)} ➜ {fmt(target['credit'])} (±{fmt(amt)})"
        ))
        return
    
    
    
    

    # ==== ลูกค้า: สมัคร/การ์ด/บัญชี/กต ====
    # ==== ลูกค้า: สมัคร/การ์ด/บัญชี/กต ====
    if text.lower() == "add":
        if uid in users:
            safe_reply(event, TextSendMessage(f"คุณมี ID แล้ว: {users[uid]['cid']}"))
            return

        name, pic = get_profile_display(event.source, uid)
        users[uid] = {
            "uid": uid,
            "cid": nextCustomerId,
            "name": name,
            "pictureUrl": pic,
            "credit": 0
        }
        nextCustomerId += 1

        safe_reply(event, flex_register_success(users[uid]["cid"]))
        return

    if text.lower() == "c":
        u = users.get(uid)
        if not u:
            safe_reply(event, TextSendMessage("พิมพ์ add เพื่อรับไอดีก่อน"))
            return
        safe_reply(event, flex_customer_card(st, u))
        return

    if text == "บช":
        safe_reply(event, flex_bank())
        return

    if text == "กต":
        safe_reply(event, TextSendMessage(rules_text()))
        return

    if text.lower() == "c":
        u = users.get(uid)
        if not u:
            # ถ้ายังไม่ได้ add → ให้ไปพิมพ์ add ก่อน
            safe_reply(event, TextSendMessage("พิมพ์ add เพื่อรับไอดีก่อน"))
            return
        safe_reply(event, flex_customer_card(st, u))
        return


    if text == "บช":
        # ✅ Flex แสดงเลขบัญชี
        safe_reply(event, flex_bank())
        return


    if text == "กต":
        # ✅ ส่งกติกาแบบข้อความธรรมดา
        safe_reply(event, TextSendMessage(rules_text()))
        return


    # ==== เปิดรอบ (O) ====
    if re.match(r"^\s*o\b", text, re.IGNORECASE):
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        note = (re.match(r"^\s*o\b\s*(.*)$", text, re.IGNORECASE).group(1) or "").strip()
        if st["phase"] == "NONE":
            st["pairNo"] += 1; st["totals"] = {"HI": 0, "LO": 0}; st["bet_index"] = {}; st["pendingCode"] = None
        st["phase"] = "OPEN"; st["note"] = note or st.get("note")
        safe_reply(event, flex_open(st["pairNo"], st.get("note"))); return

    # ==== พักรอบ (E) ====
    if text.upper() == "E":
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); return
        if st["phase"] == "NONE":
            safe_reply(event, TextSendMessage("ไม่มีรอบที่เปิดอยู่")); return
        st["phase"] = "PAUSED"
        safe_reply(event, flex_summary(st)); return

        # ==== ตั้งผล s... ====
    sm = re.match(r"^s(.+)$", text, re.IGNORECASE)
    if sm:
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน")); 
            return

        if st["phase"] != "PAUSED":   # 👈 ต้องพักรอบก่อน
            safe_reply(event, TextSendMessage("❌ ต้องกด E (พักรอบ) ก่อนจึงจะตั้งผลได้"))
            return

        st["pendingCode"] = sm.group(1).strip()
        safe_reply(event, flex_result_preview(st["pendingCode"], st["pairNo"]))
        return

             # ==== ยืนยันผล: T/ หรือ /y ====
    if text.upper() == "T/" or text.strip().lower() == "/y":
        if not is_admin(uid):
            safe_reply(event, TextSendMessage("คำสั่งนี้ใช้ได้เฉพาะแอดมิน"))
            return

        # ✅ ต้องกด E ก่อนถึงจะสรุปผลได้
        if st["phase"] != "PAUSED":
            safe_reply(event, TextSendMessage("❌ ต้องกด E (พักรอบ) ก่อนจึงจะสามารถสรุปผลได้"))
            return

        code = st["pendingCode"] or "ม"

        # 🔒 ถ้าโค้ดไม่ถูกต้อง → ห้ามสรุป
        if code not in RESULT_DEFS:
            safe_reply(event, TextSendMessage("❌ ไม่สามารถยืนยันผลได้: โค้ดผลไม่ถูกต้อง"))
            return

        # คำนวณผล
        rows, footer = settle_by_code(st, code)

        # อัปเดตเครดิตผู้เล่น
        for r in rows:
            u = users.get(r["uid"])
            if u:
                u["credit"] = max(u.get("credit", 0) + r["pnl"], 0)

        # ✅ รีเซ็ตรอบใหม่
        st["phase"] = "NONE"
        st["pendingCode"] = None
        st["bet_index"].clear()
        st["totals"] = {"HI": 0, "LO": 0}

        # คำนวณกำไรสุทธิ
        profit = -sum(r["pnl"] for r in rows)
        balance_map = {r["uid"]: users.get(r["uid"], {}).get("credit", 0) for r in rows}

        # ส่งสรุปให้ห้องปกติ
        safe_reply(event, flex_settle(st["pairNo"], rows, footer, show_profit=False, balance_map=balance_map))

        # ส่งสรุปให้ห้องหลังบ้าน (มี "กำไรสุทธิ")
        for gid_to in BACKOFFICE_GROUP_IDS:
            try:
                line_bot_api.push_message(
                    gid_to,
                    flex_settle(
                        st["pairNo"], rows, footer,
                        show_profit=True, profit_value=profit,
                        balance_map=balance_map
                    )
                )
            except Exception:
                pass
        return

    # ==== call (หลังบ้านเท่านั้น) ====
    if text.lower() == "call":
        if not gid or not is_backoffice_group_id(gid):
            return
        table = []
        for u in sorted(users.values(), key=lambda x: x["cid"]):
            fund_here = rooms[key]["funds"].get(u["uid"], 0)
            table.append({"cid": u["cid"], "name": u["name"], "credit": u.get("credit", 0), "fund": fund_here})
        pages = flex_call_pages(table)
        if not pages: pages = [TextSendMessage("ยังไม่มีลูกค้า")]

                pass

    # ==== ยกเลิกบิล ====
    m_cancel_by_cid = re.match(r"^x\s+(\d+)$", text.strip(), re.IGNORECASE)

        if st["phase"] == "NONE":
            safe_reply(event, TextSendMessage("ยกเลิกไม่ได้: รอบนี้สรุปจบแล้ว")); return

        if m_cancel_by_cid or text.strip().lower() in ("xx","x*"):
            if not is_admin(uid):

                extra = " (กำลังพักรอบ)" if st["phase"] == "PAUSED" else ""
                safe_reply(event, TextSendMessage(f"ยกเลิกบิลทั้งหมดสำเร็จ{extra} ({n} บิล)")); return
            cid = int(m_cancel_by_cid.group(1)); target = get_user_by_cid(cid)
            if not target:
                safe_reply(event, TextSendMessage(f"ไม่พบ ID {cid}")); return
            tuid = target["uid"]; bet = st["bet_index"].pop(tuid, None)
            if not bet:
                safe_reply(event, TextSendMessage(f"ID {cid} ไม่มีบิลในรอบนี้")); return
            st["totals"][bet["side"]] -= bet["amount"]
            extra = " (กำลังพักรอบ)" if st["phase"] == "PAUSED" else ""
            safe_reply(event, TextSendMessage(
                f"ยกเลิกบิลของ ID {cid} สำเร็จ{extra} ({'สูง' if bet['side']=='HI' else 'ต่ำ'} {fmt(bet['amount'])})"
            ))
            return

        if text.strip().upper() == "X":
            if st["phase"] == "PAUSED":
                safe_reply(event, TextSendMessage("กำลังพักรอบ: ลูกค้าไม่สามารถยกเลิกได้ โปรดให้แอดมินดำเนินการ"))

            bet = st["bet_index"].pop(uid, None)

                f"ยกเลิกการเดิมพันเดิมสำเร็จ ({'สูง' if bet['side']=='HI' else 'ต่ำ'} {fmt(bet['amount'])})"
            ))

    # ==== วางบิล ====
    bet = parse_bet(text)

        return

def _member_name(gid, uid):
    try:
        p = line_bot_api.get_group_member_profile(gid, uid)
        return p.display_name
    except Exception:
        return uid

def _lockdown_and_alert(gid: str, left_uid: str):
    LOCKDOWN_UNTIL[gid] = _now() + LOCKDOWN_SECONDS_DEFAULT
    who = _member_name(gid, left_uid)
    try:
        safe_push(gid, TextSendMessage(
            f"🚨 ระบบรักษาความปลอดภัย: ตรวจพบการออกจากกลุ่มของสมาชิกที่อยู่ในบัญชีเฝ้าระวัง ({who}).\n"
            f"กลุ่มถูกล็อกดาวน์ชั่วคราว {LOCKDOWN_SECONDS_DEFAULT} วินาที.\n"
            f"ให้แอดมินพิมพ์: unlock <PIN> เพื่อปลดล็อก และเชิญสมาชิกกลับเข้ากลุ่ม."
        ))
    except Exception:
        pass
    for gid_to in BACKOFFICE_GROUP_IDS:
        try:
            safe_push(gid_to, TextSendMessage(
                f"[ALERT] Protected member left in group {gid} → {who}. Lockdown engaged."))
        except Exception:
            pass

@handler.add(MemberLeftEvent)
def on_member_left(event: MemberLeftEvent):
    gid = getattr(event.source, "group_id", None)
    if not gid: return
    try:
        members = getattr(getattr(event, "left", None), "members", []) or []
    except Exception:
        members = []
    left_uids = []
    for m in members:
        uid = getattr(m, "user_id", None) or getattr(m, "mid", None) or getattr(m, "id", None)
        if uid: left_uids.append(uid)
    for lu in left_uids:
        if lu in PROTECTED_UIDS:
            _lockdown_and_alert(gid, lu)
            break

@handler.add(MemberJoinedEvent)
def on_member_joined(event: MemberJoinedEvent):
    gid = getattr(event.source, "group_id", None)
    if not gid: return
    try:
        members = getattr(getattr(event, "joined", None), "members", []) or []
    except Exception:
        members = []
    for m in members:
        uid = getattr(m, "user_id", None) or getattr(m, "mid", None) or getattr(m, "id", None)
        if uid and uid in PROTECTED_UIDS:
            LOCKDOWN_UNTIL.pop(gid, None)
            try:
                safe_push(gid, TextSendMessage("✅ สมาชิกที่เฝ้าระวังกลับเข้ากลุ่มแล้ว — ปลดล็อกโดยอัตโนมัติ"))
            except Exception:
                pass
            break
